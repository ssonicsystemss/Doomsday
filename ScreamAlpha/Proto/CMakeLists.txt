# Offline prototype of the 5CR34M 4LPH4 engine (no plugin yet): cmake -S ScreamAlpha/Proto -B <build> -G Ninja
cmake_minimum_required(VERSION 3.22)
project(AlphaProto VERSION 0.1.0)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../JUCE/CMakeLists.txt")
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../JUCE" JUCE)
else()
    include(FetchContent)
    FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(JUCE)
endif()
juce_add_console_app(AlphaRender PRODUCT_NAME "AlphaRender")
target_sources(AlphaRender PRIVATE AlphaRender.cpp)
target_compile_definitions(AlphaRender PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 $<$<PLATFORM_ID:Linux>:JUCE_ALSA=0 JUCE_JACK=0>)
target_link_libraries(AlphaRender PRIVATE juce::juce_dsp juce::juce_audio_formats juce::juce_recommended_config_flags)
